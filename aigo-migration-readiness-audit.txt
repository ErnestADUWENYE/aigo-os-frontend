  const protocol = "https:";
  const baseUrl = protocol + String.fromCharCode(47, 47) + "aigo-os.com";

  return routes.map((route) => ({
    url: `${baseUrl}${route}`,
    lastModified: new Date(),
    changeFrequency: route === "" ? "weekly" : "monthly",
    priority:
      route === ""
        ? 1
        : route === "/product" ||
            route === "/solutions" ||
            route === "/design-partners"
          ? 0.9
          : 0.7,
  }));
}
